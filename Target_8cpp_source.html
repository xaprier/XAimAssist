9double vectorLength(const std::array<double, 3>& vector) {
10 return std::sqrt(vector[0] * vector[0] + vector[1] * vector[1] +
11 vector[2] * vector[2]);
12}
13
14std::array<double, 3> normalizeOrDefault(const std::array<double, 3>& vector) {
15 const double length = vectorLength(vector);
16 if (length <= 0.000001) {
17 return {1.0, 0.0, 0.0};
18 }
19
20 return {vector[0] / length, vector[1] / length, vector[2] / length};
21}

108MovingTarget::MovingTarget(std::uint64_t TargetId, std::uint64_t SessionId,
109 SceneObjectId sceneObjId,
110 TargetDefinition definition,
111 std::chrono::steady_clock::time_point SpawnedAt)
112 : Target(TargetId, SessionId, sceneObjId, std::move(definition), SpawnedAt),
113 m_anchorPosition(CurrentPosition()) {
114 const auto& velocity = this->Definition().movementVelocity;
115 m_speed = std::max(0.1, vectorLength(velocity));
116 m_axis = normalizeOrDefault(velocity);
117 m_extent = std::max(0.1, this->Definition().movementExtent);

120void MovingTarget::Update(double dtSeconds,
121 ISceneCommandSink& sceneCommandSink) {
122 const double safeDelta = std::max(0.0, dtSeconds);
123
124 m_offset += m_direction * m_speed * safeDelta;
125 if (m_offset > m_extent) {
126 m_offset = m_extent;
127 m_direction = -1.0;
128 } else if (m_offset < -m_extent) {
129 m_offset = -m_extent;
130 m_direction = 1.0;
131 }
132
133 std::array<double, 3> nextPosition{
134 m_anchorPosition[0] + m_axis[0] * m_offset,
135 m_anchorPosition[1] + m_axis[1] * m_offset,
136 m_anchorPosition[2] + m_axis[2] * m_offset,
137 };
138
139 ApplyPosition(sceneCommandSink, nextPosition);
140}

153void StrafingTarget::Update(double dtSeconds,
154 ISceneCommandSink& sceneCommandSink) {
155 const double safeDelta = std::max(0.0, dtSeconds);
156 m_elapsedSeconds += safeDelta;
157
158 const double offset = std::sin(m_elapsedSeconds * m_angularSpeed) * m_extent;

178void BlinkingTarget::Update(double dtSeconds,
179 ISceneCommandSink& sceneCommandSink) {
180 const double safeDelta = std::max(0.0, dtSeconds);
181 m_phaseSeconds += safeDelta;
182
183 if (m_visible) {
184 if (m_phaseSeconds < m_visibleSeconds) {

193

196 }
