212void UiViewModel::UpdateFrameTick(const core::events::FrameTickEvent& event) {
213 if (event.frameSeconds <= 0.0 || !m_settings.ui.fpsCounter.enabled) {
214 return;
215 }
216
217 m_fpsAccumulatorSeconds += event.frameSeconds;
218 m_fpsAccumulatorFrames += 1;
219
220 if (m_fpsAccumulatorSeconds < 0.2) {
221 return;
222 }
223
224 const double fps =
225 static_cast<double>(m_fpsAccumulatorFrames) / m_fpsAccumulatorSeconds;
226 m_fpsAccumulatorSeconds = 0.0;
227 m_fpsAccumulatorFrames = 0;
228
229 if (_NearlyEqual(fps, m_currentFps)) {
230 return;
231 }
232
233 m_currentFps = fps;
234 emit fpsChanged();
235}

377 std::uint64_t sessionId, const std::string& overallCategory,
378 std::uint64_t modeSessionCount,
379 const std::vector<PerformanceMetricComparison>& metricComparisons) {
380 if (m_latestResult.isEmpty()) {

382 }
383
384 if (!m_latestResult.value(QStringLiteral("wasSaved")).toBool()) {
385 return;
386 }
387
388 const auto latestSessionId = static_cast<std::uint64_t>(
389 m_latestResult.value(QStringLiteral("sessionId")).toLongLong());
390 if (latestSessionId == 0 || latestSessionId != sessionId) {
391 return;
392 }
393
394 bool changed = false;
395 const auto setField = [&](const QString& key, const QVariant& value) {
396 if (m_latestResult.value(key) == value) {
397 return;
398 }
399
400 m_latestResult.insert(key, value);
401 changed = true;
402 };
403
404 QVariantList comparisons;
405 comparisons.reserve(static_cast<qsizetype>(metricComparisons.size()));
406 for (const auto& comparison : metricComparisons) {
407 QVariantMap comparisonVariant;

409 QString::fromStdString(comparison.metricId));

411 QString::fromStdString(comparison.tier));

413 comparisonVariant.insert(QStringLiteral("average"), comparison.average);
414 comparisonVariant.insert(QStringLiteral("best"), comparison.best);
415 comparisonVariant.insert(QStringLiteral("worst"), comparison.worst);
416 comparisons.push_back(comparisonVariant);
417 }
418
419 setField(QStringLiteral("performanceTier"),
420 QString::fromStdString(overallCategory));
421 setField(QStringLiteral("wasSaved"), true);
422 setField(QStringLiteral("saveState"), QStringLiteral("saved"));
423 setField(QStringLiteral("invalidReasonCode"), QStringLiteral(""));
424 setField(QStringLiteral("modeSessionCount"),

1275void UiViewModel::SetModeSettingValue(const QString& settingKey, double value) {
1276 const ModeDescriptor* mode = _SelectedModeDescriptor();
1277 if (mode == nullptr) {
1278 return;
1279 }
1280
1281 const std::string key = settingKey.toStdString();
1282 const auto settingIterator = std::find_if(
1283 mode->settings.begin(), mode->settings.end(),
1284 [&](const ModeSettingDescriptor& setting) { return setting.key == key; });
1285
1286 if (settingIterator == mode->settings.end()) {
1287 return;
1288 }
1289
1290 const double low =
1291 std::min(settingIterator->minValue, settingIterator->maxValue);
1292 const double high =
1293 std::max(settingIterator->minValue, settingIterator->maxValue);
1294
1295 double clampedValue = std::clamp(value, low, high);
1296 if (settingIterator->integerOnly) {
1297 clampedValue = std::round(clampedValue);
1298 }
1299
1300 m_modeSettingValues[key] = clampedValue;
1301
1302 if (key == "grid_rows") {
1303 m_modeGridRows = static_cast<int>(clampedValue);
1304 } else if (key == "grid_columns") {
1305 m_modeGridColumns = static_cast<int>(clampedValue);
1306 } else if (key == "active_target_count") {
1307 m_modeActiveTargetCount = static_cast<int>(clampedValue);
1308 }
1309
1310 if (_SelectedModeSupportsGridConfig()) {
1311 const int clampedActiveCount = _ClampedActiveTargetCountForGrid(
1312 m_modeActiveTargetCount, m_modeGridRows, m_modeGridColumns);
1313 m_modeActiveTargetCount = clampedActiveCount;
1314 m_modeSettingValues["active_target_count"] =
1315 static_cast<double>(clampedActiveCount);
1316 }
1317
1318 _RebuildModeSettingsVariant();
1319 emit modeSettingsChanged();
1320
1321 _PersistSelectedModePreferences();
1322 _EmitSettingsApplied();
1323}

1753 if (std::isfinite(modePreferences.durationSeconds) &&
