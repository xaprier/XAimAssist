9bool RenderContext::Initialize(QVTKOpenGLNativeWidget& viewportWidget) {
10 if (m_renderer != nullptr && m_renderWindow != nullptr) {
11 return true;
12 }
13
14 m_viewportWidget = &viewportWidget;
15 m_renderWindow = vtkSmartPointer<vtkGenericOpenGLRenderWindow>::New();
16 m_renderer = vtkSmartPointer<vtkRenderer>::New();
17
18 // Disable VSync: for a competitive aim trainer, latency matters more than
19 // tear-free presentation. swap_interval=0 lets the GPU render as fast as
20 // possible and is the standard setting in competitive FPS applications.
21 m_renderWindow->SetSwapControl(0);
22
23 // Enable double buffering (front + one back buffer). Triple-buffering
24 // is not requested; driver-side triple-buffering is disabled via VSync=off.
25 m_renderWindow->DoubleBufferOn();
26
27 // Disable VTK-level multisampling; QSurfaceFormat also sets samples=0.
28 // Both must be zero to avoid the driver silently enabling MSAA on one path.
29 m_renderWindow->SetMultiSamples(0);
30
31 m_renderWindow->AddRenderer(m_renderer);
32 m_viewportWidget->setRenderWindow(m_renderWindow);
33
34 return true;
35}
