23QString serializeModeOverridesJson(
24 const std::unordered_map<
25 std::string, xaimassist::persistence::ModePreferences>& modeOverrides) {
26 QJsonObject root;
27
28 for (const auto& [modeId, modePreferences] : modeOverrides) {
29 if (modeId.empty()) {
30 continue;
31 }
32
33 QJsonObject modeObject;
34
35 if (std::isfinite(modePreferences.durationSeconds) &&
36 modePreferences.durationSeconds > 0.0) {
37 modeObject.insert(QStringLiteral("durationSeconds"),
38 modePreferences.durationSeconds);
39 }
40
41 if (std::isfinite(modePreferences.distanceUnits) &&
42 modePreferences.distanceUnits > 0.0) {
43 modeObject.insert(QStringLiteral("distanceUnits"),
44 modePreferences.distanceUnits);
45 }
46
47 QJsonObject settingValuesObject;
48 for (const auto& [settingKey, settingValue] :
49 modePreferences.settingValues) {
50 if (settingKey.empty() || !std::isfinite(settingValue)) {
51 continue;
52 }
53
54 settingValuesObject.insert(QString::fromStdString(settingKey),
55 settingValue);
56 }
57
58 modeObject.insert(QStringLiteral("settingValues"), settingValuesObject);
59 root.insert(QString::fromStdString(modeId), modeObject);
60 }
61
62 return QString::fromUtf8(QJsonDocument(root).toJson(QJsonDocument::Compact));
63}

65std::unordered_map<std::string, xaimassist::persistence::ModePreferences>
66parseModeOverridesJson(const QString& jsonValue) {
67 std::unordered_map<std::string, xaimassist::persistence::ModePreferences>
68 modeOverrides;
69
70 const QByteArray payload = jsonValue.trimmed().toUtf8();
71 if (payload.isEmpty()) {
72 return modeOverrides;
73 }
74
75 QJsonParseError parseError;
76 const QJsonDocument document = QJsonDocument::fromJson(payload, &parseError);
77 if (parseError.error != QJsonParseError::NoError || !document.isObject()) {
78 return modeOverrides;
79 }
80
81 const QJsonObject root = document.object();
82 for (auto modeIterator = root.begin(); modeIterator != root.end();
83 ++modeIterator) {
84 if (!modeIterator.value().isObject()) {
85 continue;
86 }
87
88 const std::string modeId = modeIterator.key().trimmed().toStdString();
89 if (modeId.empty()) {
90 continue;
91 }
92
93 xaimassist::persistence::ModePreferences modePreferences;
94 const QJsonObject modeObject = modeIterator.value().toObject();
95
96 const double durationSeconds =
97 modeObject.value(QStringLiteral("durationSeconds")).toDouble(0.0);
98 if (std::isfinite(durationSeconds) && durationSeconds > 0.0) {
99 modePreferences.durationSeconds = durationSeconds;
100 }
101
102 const double distanceUnits =
103 modeObject.value(QStringLiteral("distanceUnits")).toDouble(0.0);
104 if (std::isfinite(distanceUnits) && distanceUnits > 0.0) {
105 modePreferences.distanceUnits = distanceUnits;
106 }
107
108 const QJsonValue settingValuesValue =
109 modeObject.value(QStringLiteral("settingValues"));
110 if (settingValuesValue.isObject()) {
111 const QJsonObject settingValuesObject = settingValuesValue.toObject();
112 for (auto settingIterator = settingValuesObject.begin();
113 settingIterator != settingValuesObject.end(); ++settingIterator) {
114 if (!settingIterator.value().isDouble()) {
115 continue;
116 }
117
118 const std::string settingKey =
119 settingIterator.key().trimmed().toStdString();
120 const double settingValue = settingIterator.value().toDouble();
121 if (settingKey.empty() || !std::isfinite(settingValue)) {
122 continue;
123 }
124
125 modePreferences.settingValues.emplace(settingKey, settingValue);
126 }
127 }
128
129 modeOverrides.emplace(modeId, std::move(modePreferences));
130 }
131
132 return modeOverrides;
133}
