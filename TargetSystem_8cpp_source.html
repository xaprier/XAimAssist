17double toReactionMs(std::chrono::steady_clock::time_point SpawnedAt,
18 std::chrono::steady_clock::time_point shotAt) {
19 if (shotAt <= SpawnedAt) {
20 return 0.0;
21 }
22
23 return std::chrono::duration<double, std::milli>(shotAt - SpawnedAt).count();
24}

41TargetSystem::TargetId TargetSystem::SpawnTarget(std::uint64_t SessionId,
42 const TargetDefinition& Definition) {
43 if (SessionId == 0) {
44 return 0;
45 }
46
47 SphereTargetSpawnRequest spawnRequest;
48 spawnRequest.radius = Definition.radius;
49 spawnRequest.position = Definition.initialPosition;
50 spawnRequest.color = Definition.color;
51 spawnRequest.collidable = Definition.collidable;
52
53 const SceneObjectId sceneObjectId =
54 m_sceneCommandSink.SpawnSphereTarget(spawnRequest);
55 if (sceneObjectId == 0) {
56 m_logger.Error(
57 "gameplay",
58 "Target spawn failed: scene object creation returned invalid id");
59 return 0;
60 }
61
62 const auto SpawnedAt = std::chrono::steady_clock::now();
63 const TargetId targetId = static_cast<TargetId>(sceneObjectId);
64
65 auto target = m_targetFactory.CreateTarget(targetId, SessionId, sceneObjectId,
66 Definition, SpawnedAt);
67 if (!target) {
68 m_sceneCommandSink.DestroyTarget(sceneObjectId);
69 m_logger.Error(
70 "gameplay",
71 "Target spawn failed: no target implementation for Definition");
72 return 0;
73 }
74
75 m_targets.emplace(targetId, std::move(target));
76
77 m_eventBus.Publish(
78 core::events::TargetSpawnedEvent{SessionId, targetId, SpawnedAt});
79
80 return targetId;
81}

104void TargetSystem::ClearSessionTargets(std::uint64_t SessionId) {
105 if (SessionId == 0) {
106 return;
107 }
108
109 std::vector<TargetId> targetsToDestroy;
110 targetsToDestroy.reserve(m_targets.size());
111
112 for (const auto& [targetId, target] : m_targets) {
113 if (target->SessionId() == SessionId) {
114 targetsToDestroy.push_back(targetId);
115 }
116 }
117
118 const auto timestamp = std::chrono::steady_clock::now();
119 for (const auto targetId : targetsToDestroy) {
120 const auto iterator = m_targets.find(targetId);
121 if (iterator == m_targets.end()) {
122 continue;
123 }
124
125 m_sceneCommandSink.DestroyTarget(iterator->second->GetSceneObjectId());
126
127 m_eventBus.Publish(core::events::TargetDestroyedEvent{SessionId, targetId,
128 false, timestamp});
129
130 m_targets.erase(iterator);
131 }

190void TargetSystem::_OnShotFired(
191 const core::events::ShotFiredEvent& shotFiredEvent) {
192 const auto hitResult = m_hitTestService.CastShotRay(shotFiredEvent);
193 if (!hitResult.has_value()) {
194 m_eventBus.Publish(core::events::ShotMissEvent{shotFiredEvent.sessionId,
195 shotFiredEvent.timestamp});
196 return;
197 }
198
199 const TargetId targetId = hitResult->TargetId;
200 const auto iterator = m_targets.find(targetId);
201 if (iterator == m_targets.end()) {
202 m_eventBus.Publish(core::events::ShotMissEvent{shotFiredEvent.sessionId,
203 shotFiredEvent.timestamp});
204 return;
205 }
206
207 auto& target = *iterator->second;
208 if (target.SessionId() != shotFiredEvent.sessionId ||
209 target.State() != TargetState::Alive) {
210 m_eventBus.Publish(core::events::ShotMissEvent{shotFiredEvent.sessionId,
211 shotFiredEvent.timestamp});
212 return;
213 }
214
215 const double reactionTimeMs =
216 toReactionMs(target.SpawnedAt(), shotFiredEvent.timestamp);
217 target.OnHit(m_sceneCommandSink);
218
219 m_eventBus.Publish(core::events::ShotHitEvent{shotFiredEvent.sessionId,
220 targetId, reactionTimeMs,
221 shotFiredEvent.timestamp});
222
223 if (target.State() == TargetState::Destroyed) {
224 m_eventBus.Publish(core::events::TargetDestroyedEvent{
225 shotFiredEvent.sessionId, targetId, true, shotFiredEvent.timestamp});
226 m_targets.erase(iterator);
227 }
228}
