11void WorldRenderSync::Sync() {
12 const std::uint64_t currentTopologyRevision =
13 m_world.RenderTopologyRevision();
14 const bool topologyChanged =
15 currentTopologyRevision != m_lastTopologyRevision;
16 const bool hasDirtyEntities = m_world.RenderDirtyCount() > 0;
17
18 // Fast-path: skip sync when nothing changed
19 if (!topologyChanged && !hasDirtyEntities) {
20 return;
21 }
22
23 auto syncEntity = [&](EntityId entityId) {
24 const auto* transformComponent = m_world.Transform(entityId);
25 const auto* renderComponent = m_world.Render(entityId);
26 if (transformComponent == nullptr || renderComponent == nullptr ||
27 !renderComponent->visible) {
28 m_world.ClearRenderDirty(entityId);
29 return;
30 }
31
32 auto bindingIterator = m_renderBindings.find(entityId);
33 if (bindingIterator == m_renderBindings.end()) {
34 IRenderSceneBackend::RenderObjectId renderObjectId = 0;
35 switch (renderComponent->primitiveType) {
36 case RenderPrimitiveType::Sphere:
37 renderObjectId =
38 m_renderBackend.CreateSphere(*transformComponent, *renderComponent);
39 break;
40 }
41
42 if (renderObjectId != 0) {
43 m_renderBindings.emplace(entityId, renderObjectId);
44 m_world.ClearRenderDirty(entityId);
45 }
46
47 return;
48 }
49
50 if (!m_world.IsRenderDirty(entityId)) {
51 return;
52 }
53
54 bool updated = false;
55 switch (renderComponent->primitiveType) {
56 case RenderPrimitiveType::Sphere:
57 updated = m_renderBackend.UpdateSphere(
58 bindingIterator->second, *transformComponent, *renderComponent);
59 break;
60 }
61
62 if (updated) {
63 m_world.ClearRenderDirty(entityId);
64 }
65 };
66
67 if (topologyChanged) {
68 // Prune render bindings for destroyed entities
69 const auto renderableEntities = m_world.EntitiesWithRenderable();
70 const std::unordered_set<EntityId> activeEntities(
71 renderableEntities.begin(), renderableEntities.end());
72
73 for (auto iterator = m_renderBindings.begin();
74 iterator != m_renderBindings.end();) {
75 if (activeEntities.find(iterator->first) != activeEntities.end()) {
76 ++iterator;
77 continue;
78 }
79
80 m_renderBackend.DestroyObject(iterator->second);
81 iterator = m_renderBindings.erase(iterator);
82 }
83
84 for (const EntityId entityId : renderableEntities) {
85 syncEntity(entityId);
86 }
87
88 m_lastTopologyRevision = currentTopologyRevision;
89 return;
90 }
91
92 const auto dirtyEntities = m_world.RenderDirtyEntities();
93 for (const EntityId entityId : dirtyEntities) {
94 syncEntity(entityId);
95 }
96}
