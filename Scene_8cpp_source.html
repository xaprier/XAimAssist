26std::array<double, 3> _Mix(const std::array<double, 3>& left,
27 const std::array<double, 3>& right, double factor) {
28 const double clampedFactor = std::clamp(factor, 0.0, 1.0);
29 return {
30 left[0] + (right[0] - left[0]) * clampedFactor,
31 left[1] + (right[1] - left[1]) * clampedFactor,
32 left[2] + (right[2] - left[2]) * clampedFactor,
33 };
34}

58vtkSmartPointer<vtkActor> createLineActor(const std::array<double, 3>& Start,
59 const std::array<double, 3>& end,
60 const std::array<double, 3>& color,
61 double lineWidth, double opacity) {
62 auto lineSource = vtkSmartPointer<vtkLineSource>::New();
63 lineSource->SetPoint1(Start[0], Start[1], Start[2]);
64 lineSource->SetPoint2(end[0], end[1], end[2]);
65
66 auto mapper = vtkSmartPointer<vtkPolyDataMapper>::New();
67 mapper->SetInputConnection(lineSource->GetOutputPort());
68
69 auto actor = vtkSmartPointer<vtkActor>::New();
70 actor->SetMapper(mapper);
71 actor->PickableOff();
72 actor->GetProperty()->SetColor(color[0], color[1], color[2]);
73 actor->GetProperty()->SetLineWidth(lineWidth);
74 actor->GetProperty()->SetOpacity(opacity);
75
76 return actor;

234void Scene::_RebuildEnvironment() {
235 _ClearEnvironmentActors();
236
237 if (m_renderer == nullptr) {
238 return;
239 }

241 const auto referenceColor =
242 referenceColorForBackground(m_environmentBackgroundColor);
243
244 const auto floorMinorColor =
245 _Mix(m_environmentBackgroundColor, referenceColor, 0.20);
246 const auto floorMajorColor =
247 _Mix(m_environmentBackgroundColor, referenceColor, 0.36);
248 const auto wallMinorColor =
249 _Mix(m_environmentBackgroundColor, referenceColor, 0.24);
250 const auto wallMajorColor =

253 constexpr double GRID_SPACING = 10.0;
254 constexpr int GRID_MAJOR_STEP = 5;
255
256 const double minX = SCENE_MIN_BOUNDS[0];
257 const double minY = SCENE_MIN_BOUNDS[1];
258 const double minZ = SCENE_MIN_BOUNDS[2];
259 const double maxX = SCENE_MAX_BOUNDS[0];
260 const double maxY = SCENE_MAX_BOUNDS[1];
261 const double maxZ = SCENE_MAX_BOUNDS[2];
262
263 const int xSteps = static_cast<int>((maxX - minX) / GRID_SPACING);
264 const int ySteps = static_cast<int>((maxY - minY) / GRID_SPACING);
265 const int zSteps = static_cast<int>((maxZ - minZ) / GRID_SPACING);
266
267 auto addGridLine = [&](const std::array<double, 3>& Start,
268 const std::array<double, 3>& end, bool isMajor,
269 bool floorLikeSurface) {
270 const auto& lineColor = floorLikeSurface
271 ? (isMajor ? floorMajorColor : floorMinorColor)
272 : (isMajor ? wallMajorColor : wallMinorColor);
273 const double lineWidth = isMajor ? 1.8 : 1.0;
274 const double opacity =
275 floorLikeSurface ? (isMajor ? 0.62 : 0.36) : (isMajor ? 0.56 : 0.32);
276 _AddEnvironmentActor(
277 createLineActor(Start, end, lineColor, lineWidth, opacity));
278 };
279
280 for (int ix = 0; ix <= xSteps; ++ix) {
281 const double x = minX + static_cast<double>(ix) * GRID_SPACING;
282 const bool isMajorLine = (ix % GRID_MAJOR_STEP) == 0;
283
284 addGridLine({x, minY, minZ}, {x, minY, maxZ}, isMajorLine, true);
285 addGridLine({x, maxY, minZ}, {x, maxY, maxZ}, isMajorLine, true);
286 addGridLine({x, minY, minZ}, {x, maxY, minZ}, isMajorLine, false);
287 addGridLine({x, minY, maxZ}, {x, maxY, maxZ}, isMajorLine, false);
288 }
289
290 for (int iz = 0; iz <= zSteps; ++iz) {
291 const double z = minZ + static_cast<double>(iz) * GRID_SPACING;
292 const bool isMajorLine = (iz % GRID_MAJOR_STEP) == 0;
293
294 addGridLine({minX, minY, z}, {maxX, minY, z}, isMajorLine, true);
295 addGridLine({minX, maxY, z}, {maxX, maxY, z}, isMajorLine, true);
296 addGridLine({minX, minY, z}, {minX, maxY, z}, isMajorLine, false);
297 addGridLine({maxX, minY, z}, {maxX, maxY, z}, isMajorLine, false);
298 }
299
300 for (int iy = 0; iy <= ySteps; ++iy) {
301 const double y = minY + static_cast<double>(iy) * GRID_SPACING;
302 const bool isMajorLine = (iy % GRID_MAJOR_STEP) == 0;
303
304 addGridLine({minX, y, minZ}, {maxX, y, minZ}, isMajorLine, false);
305 addGridLine({minX, y, maxZ}, {maxX, y, maxZ}, isMajorLine, false);
306
307 const auto& sideHorizontalColor =
308 isMajorLine ? wallMajorColor : wallMinorColor;
309 _AddEnvironmentActor(
310 createLineActor({minX, y, minZ}, {minX, y, maxZ}, sideHorizontalColor,
311 isMajorLine ? 2.2 : 1.5, isMajorLine ? 0.74 : 0.54));
312 _AddEnvironmentActor(
313 createLineActor({maxX, y, minZ}, {maxX, y, maxZ}, sideHorizontalColor,
314 isMajorLine ? 2.2 : 1.5, isMajorLine ? 0.74 : 0.54));
315 }
316
317 _AddEnvironmentActor(createRingActor({0.0, minY + 3.0, 0.0}, 24.0,
318 {0.0, 1.0, 0.0}, wallMajorColor, 1.4,
319 0.35));
320}
