18QString I18nProvider::ModeName(persistence::UiLanguage language,
19 const std::string& modeId,
20 const std::string& fallback) {
21 if (language == persistence::UiLanguage::Turkish) {
22 if (modeId == "static_sphere") return QStringLiteral("Sabit Küre");
23 if (modeId == "gridshot") return QStringLiteral("GridShot");
24 if (modeId == "next_shot") return QStringLiteral("Next Shot");
25 if (modeId == "strafing_targets") return QStringLiteral("Strafing Hedefler");
26 if (modeId == "tracking_targets") return QStringLiteral("Tracking Hedefler");
27 }
28
29 return QString::fromStdString(fallback);

32QString I18nProvider::ModeDescription(persistence::UiLanguage language,
33 const std::string& modeId,
34 const std::string& fallback) {
35 if (language == persistence::UiLanguage::Turkish) {
36 if (modeId == "static_sphere") {
37 return QStringLiteral(
38 "Tek bir sabit küre hedefi üretir; vurulduğunda "
39 "yeni bir konuma taşır.");
40 }
41 if (modeId == "gridshot") {
42 return QStringLiteral(
43 "Sabit bir uzaklık düzleminde varsayılan 5x5 grid ve 3 aktif hedef "
44 "ile başlar; satır/sütun/aktif hedef sayısı ayarlanabilir ve "
45 "vurulan hedef boş bir hücrede rastgele yeniden üretilir.");
46 }
47 if (modeId == "next_shot") {
48 return QStringLiteral(
49 "Static Sphere ile benzer çalışır; fark olarak bir sonraki hedefin "
50 "konumu düşük opaklıkta önizleme küresi ile önceden gösterilir.");
51 }
52 if (modeId == "strafing_targets") {
53 return QStringLiteral(
54 "Tek bir sabit küre ile başlar. İlk isabet küreyi rastgele bir YZ "
55 "ekseninde ayarlı hızda sürekli hareket ettirir. Hareketli küre 2.5 "
56 "saniye içinde vurulmazsa miss sayılır ve yeni sabit küre rastgele "
57 "konumda üretilir.");
58 }
59 if (modeId == "tracking_targets") {
60 return QStringLiteral(
61 "Tek bir hareketli küre üretir. Küre ayarlı hızda sürekli hareket "
62 "eder, YZ alan sınırına geldiğinde sekme benzeri şekilde ters yönde "
63 "rastgele eksende hareketine devam eder.");
64 }
65 }
66
67 return QString::fromStdString(fallback);
68}

70QString I18nProvider::ModeSettingName(persistence::UiLanguage language,
71 const std::string& modeId,
72 const std::string& settingKey,
73 const std::string& fallback) {
74 if (language == persistence::UiLanguage::Turkish) {
75 if (modeId == "gridshot") {
76 if (settingKey == "grid_rows") return QStringLiteral("Grid Satır");
77 if (settingKey == "grid_columns") return QStringLiteral("Grid Sütun");
78 if (settingKey == "active_target_count") return QStringLiteral("Aktif Hedef");
79 }
80 if (modeId == "strafing_targets" || modeId == "tracking_targets") {
81 if (settingKey == "speed") return QStringLiteral("Hız");
82 }
83 }
84
85 return QString::fromStdString(fallback);
86}
