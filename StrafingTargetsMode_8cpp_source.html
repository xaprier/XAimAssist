60void StrafingTargetsMode::OnStart() {
61 m_activeTargetId = 0;
62 m_previousStaticSpawnOffset.reset();
63 m_activeTargetSpawnOffset.reset();
64 m_activeTargetPosition = {0.0, 0.0, 0.0};
65 m_activeMovementDirection = {0.0, 1.0};
66 m_activeMovementSpeed = STRAFE_SPEED_DEFAULT;
67 m_movingElapsedSeconds = 0.0;
68 m_targetStage = TargetStage::StaticAwaitingActivation;
69
70 _SpawnStaticTargetAtRandomSlot();
71
72 m_shotHitSubscriptionId =
73 GetEventBus().Subscribe([this](const core::events::CoreEvent& event) {
74 const auto* shotHitEvent =
75 std::get_if<core::events::ShotHitEvent>(&event);
76 if (shotHitEvent == nullptr) {
77 return;
78 }
79
80 if (shotHitEvent->sessionId != SessionId()) {
81 return;
82 }
83
84 if (shotHitEvent->targetId != m_activeTargetId) {
85 return;
86 }
87
88 if (!m_activeTargetSpawnOffset.has_value()) {
89 m_targetStage = TargetStage::StaticAwaitingActivation;
90 m_movingElapsedSeconds = 0.0;
91 _SpawnStaticTargetAtRandomSlot();
92 return;
93 }
94
95 if (m_targetStage == TargetStage::StaticAwaitingActivation) {
96 const double speed =
97 std::max(STRAFE_SPEED_MIN,
98 ModeSettingValue("speed", STRAFE_SPEED_DEFAULT));
99
100 if (!_SpawnMovingTargetAtOffset(*m_activeTargetSpawnOffset, speed)) {
101 m_targetStage = TargetStage::StaticAwaitingActivation;
102 m_movingElapsedSeconds = 0.0;
103 _SpawnStaticTargetAtRandomSlot();
104 return;
105 }
106
107 m_targetStage = TargetStage::MovingAwaitingHit;
108 return;
109 }
110
111 m_targetStage = TargetStage::StaticAwaitingActivation;
112 m_movingElapsedSeconds = 0.0;
113 _SpawnStaticTargetAtRandomSlot();
114 });
115
116 GetLogger().Info("gameplay", "StrafingTargetsMode started");
117}

119void StrafingTargetsMode::OnUpdate(double dtSeconds) {
120 if (m_targetStage != TargetStage::MovingAwaitingHit ||
121 m_activeTargetId == 0) {
122 return;
123 }
124
125 if (!GetTargetSystem().HasTarget(m_activeTargetId)) {
126 return;
127 }
128
129 const double safeDelta = std::max(0.0, dtSeconds);
130 m_activeTargetPosition[1] +=
131 m_activeMovementDirection[0] * m_activeMovementSpeed * safeDelta;
132 m_activeTargetPosition[2] +=
133 m_activeMovementDirection[1] * m_activeMovementSpeed * safeDelta;
134
135 if (!GetSceneCommandSink().UpdateTargetPosition(
136 static_cast<SceneObjectId>(m_activeTargetId),
137 m_activeTargetPosition)) {
138 return;
139 }
140
141 m_movingElapsedSeconds += safeDelta;
142 if (m_movingElapsedSeconds < MOVING_MISS_TIMEOUT_SECONDS) {
143 return;
144 }
145
146 GetTargetSystem().ClearSessionTargets(SessionId());
147 GetEventBus().Publish(core::events::ShotMissEvent{
148 SessionId(), std::chrono::steady_clock::now()});
149
150 m_targetStage = TargetStage::StaticAwaitingActivation;
151 m_movingElapsedSeconds = 0.0;
152 _SpawnStaticTargetAtRandomSlot();
153}

173void StrafingTargetsMode::_SpawnStaticTargetAtRandomSlot() {
174 const GameplayRuntimeSettings runtimeSettings =
175 GetSettingsProvider().CurrentSettings();
176
177 TargetDefinition Definition;
178 Definition.type = TargetType::Static;
179 Definition.radius = runtimeSettings.TargetRadius;
180
181 const double spawnDistanceX = std::max(0.1, ConfiguredDistance());
182 const double spawnRangeY = TARGET_Y_MAX - TARGET_Y_MIN;
183 const double spawnRangeZ = TARGET_Z_MAX - TARGET_Z_MIN;
184 const double maxSpawnOffsetDistance =
185 std::sqrt(spawnRangeY * spawnRangeY + spawnRangeZ * spawnRangeZ);
186 const double requestedMinimumSpawnOffsetDistance = spawnDistanceX / 5.0;
187 const double cappedMinimumSpawnOffsetDistance =
188 std::min(requestedMinimumSpawnOffsetDistance,
189 maxSpawnOffsetDistance * MIN_SPAWN_DISTANCE_CAP_FACTOR);
190 const double _MinimumSpawnOffsetDistance =
191 std::clamp(cappedMinimumSpawnOffsetDistance, 0.0, maxSpawnOffsetDistance);
192
193 const auto spawnOffset = _RandomSpawnOffset(_MinimumSpawnOffsetDistance);
194 Definition.initialPosition = {spawnDistanceX, spawnOffset[0], spawnOffset[1]};
195 Definition.color = runtimeSettings.targetColor;
196 Definition.collidable = true;
197
198 m_activeTargetId = GetTargetSystem().SpawnTarget(SessionId(), Definition);
199 if (m_activeTargetId == 0) {
200 m_activeTargetPosition = {0.0, 0.0, 0.0};
201 m_activeTargetSpawnOffset.reset();
202 GetLogger().Warning("gameplay",
203 "StrafingTargetsMode failed to spawn static target");
204 return;
205 }
206
207 m_targetStage = TargetStage::StaticAwaitingActivation;
208 m_movingElapsedSeconds = 0.0;
209 m_activeTargetSpawnOffset = spawnOffset;
210 m_previousStaticSpawnOffset = spawnOffset;
211 m_activeTargetPosition = Definition.initialPosition;
212}

214bool StrafingTargetsMode::_SpawnMovingTargetAtOffset(
215 const std::array<double, 2>& spawnOffset, double speed) {
216 const GameplayRuntimeSettings runtimeSettings =
217 GetSettingsProvider().CurrentSettings();
218 const double clampedSpeed =
219 std::clamp(speed, STRAFE_SPEED_MIN, STRAFE_SPEED_MAX);
220 const std::array<double, 2> direction = _RandomUnitPlanarDirection();
221
222 TargetDefinition Definition;
223 Definition.type = TargetType::Static;
224 Definition.radius = runtimeSettings.TargetRadius;
225 Definition.initialPosition = {std::max(0.1, ConfiguredDistance()),
226 spawnOffset[0], spawnOffset[1]};
227 Definition.color = runtimeSettings.targetColor;
228 Definition.collidable = true;
229
230 m_activeTargetId = GetTargetSystem().SpawnTarget(SessionId(), Definition);
231 if (m_activeTargetId == 0) {
232 GetLogger().Warning("gameplay",
233 "StrafingTargetsMode failed to spawn moving target");
234 return false;
235 }
236
237 m_activeTargetSpawnOffset = spawnOffset;
238 m_activeTargetPosition = Definition.initialPosition;
239 m_activeMovementDirection = direction;
240 m_activeMovementSpeed = clampedSpeed;
241 m_movingElapsedSeconds = 0.0;
242 return true;
243}

245std::array<double, 2>
246StrafingTargetsMode::_RandomSpawnOffset(double minimumDistance) {
247 std::uniform_real_distribution<double> yDistribution(TARGET_Y_MIN,
248 TARGET_Y_MAX);
249 std::uniform_real_distribution<double> zDistribution(TARGET_Z_MIN,
250 TARGET_Z_MAX);
251
252 std::array<double, 2> bestOffset = {
253 yDistribution(m_randomEngine),
254 zDistribution(m_randomEngine),
255 };
256
257 if (!m_previousStaticSpawnOffset.has_value()) {
258 return bestOffset;
259 }
260
261 double bestDistance =
262 _PlanarDistance(bestOffset, *m_previousStaticSpawnOffset);
263 if (bestDistance >= minimumDistance) {
264 return bestOffset;
265 }
266
267 for (std::size_t attempt = 0; attempt < MAX_RANDOM_SAMPLE_ATTEMPTS;
268 ++attempt) {
269 const std::array<double, 2> candidateOffset = {
270 yDistribution(m_randomEngine),
271 zDistribution(m_randomEngine),
272 };
273
274 const double candidateDistance =
275 _PlanarDistance(candidateOffset, *m_previousStaticSpawnOffset);
276 if (candidateDistance >= minimumDistance) {
277 return candidateOffset;
278 }
279
280 if (candidateDistance > bestDistance) {
281 bestDistance = candidateDistance;
282 bestOffset = candidateOffset;
283 }
284 }
285
286 return bestOffset;
287}

289std::array<double, 2> StrafingTargetsMode::_RandomUnitPlanarDirection() {
290 std::uniform_real_distribution<double> angleDistribution(0.0, 2.0 * PI);
291 const double angle = angleDistribution(m_randomEngine);
292 return {std::cos(angle), std::sin(angle)};
293}

295double
296StrafingTargetsMode::_PlanarDistance(const std::array<double, 2>& left,
297 const std::array<double, 2>& right) {
298 const double deltaY = left[0] - right[0];
299 const double deltaZ = left[1] - right[1];
300 return std::sqrt(deltaY * deltaY + deltaZ * deltaZ);
301}
