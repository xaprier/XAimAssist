387void TrainingFlowController::_CollectPipelineSample(std::uint64_t sessionId,
388 std::uint32_t fixedSteps,
389 double framePipelineMs,
390 double uiFrameTickMs,
391 double modeUpdateMs,
392 double worldSyncMs) {
393 if (!m_pipelineTimings.active ||
394 m_pipelineTimings.sessionId != sessionId) {
395 return;
396 }
397
398 ++m_pipelineTimings.sampledFrames;
399 m_pipelineTimings.sampledFixedSteps += fixedSteps;
400
401 m_pipelineTimings.framePipelineMsSum += framePipelineMs;
402 m_pipelineTimings.uiFrameTickMsSum += uiFrameTickMs;
403 m_pipelineTimings.modeUpdateMsSum += modeUpdateMs;
404 m_pipelineTimings.worldSyncMsSum += worldSyncMs;
405
406 m_pipelineTimings.framePipelineWorstMs =
407 std::max(m_pipelineTimings.framePipelineWorstMs, framePipelineMs);
408 m_pipelineTimings.uiFrameTickWorstMs =
409 std::max(m_pipelineTimings.uiFrameTickWorstMs, uiFrameTickMs);
410 m_pipelineTimings.modeUpdateWorstMs =
411 std::max(m_pipelineTimings.modeUpdateWorstMs, modeUpdateMs);
412 m_pipelineTimings.worldSyncWorstMs =
413 std::max(m_pipelineTimings.worldSyncWorstMs, worldSyncMs);
414}

416void TrainingFlowController::_LogPipelineReport(
417 const core::events::SessionStoppedEvent& event) {
418 if (!m_pipelineTimings.active ||
419 m_pipelineTimings.sessionId != event.sessionId) {
420 return;
421 }
422
423 if (m_pipelineTimings.sampledFrames == 0) {
424 m_deps.logger.LogIf(
425 "DEBUG_PERF", core::LogLevel::Info, "perf",
426 "Session pipeline report unavailable: no collected frames");
427 _ResetPipelineTimings();
428 return;
429 }
430
431 const double sampledFrames =
432 static_cast<double>(m_pipelineTimings.sampledFrames);
433 const double avgFramePipelineMs =
434 m_pipelineTimings.framePipelineMsSum / sampledFrames;
435 const double avgUiFrameTickMs =
436 m_pipelineTimings.uiFrameTickMsSum / sampledFrames;
437 const double avgModeUpdateMs =
438 m_pipelineTimings.modeUpdateMsSum / sampledFrames;
439 const double avgWorldSyncMs =
440 m_pipelineTimings.worldSyncMsSum / sampledFrames;
441 const double avgFixedStepsPerFrame =
442 static_cast<double>(m_pipelineTimings.sampledFixedSteps) / sampledFrames;
443 const double estimatedFps =
444 event.elapsedSeconds > 0.0
445 ? sampledFrames / event.elapsedSeconds
446 : 0.0;
447 const double avgBudget144Pct =
448 kTargetFrameTime144Ms > 0.0
449 ? (avgFramePipelineMs * 100.0 / kTargetFrameTime144Ms)
450 : 0.0;
451 const double avgBudget120Pct =
452 kTargetFrameTime120Ms > 0.0
453 ? (avgFramePipelineMs * 100.0 / kTargetFrameTime120Ms)
454 : 0.0;
455 const double worstBudget144Pct =
456 kTargetFrameTime144Ms > 0.0
457 ? (m_pipelineTimings.framePipelineWorstMs * 100.0 /
458 kTargetFrameTime144Ms)
459 : 0.0;
460
461 std::ostringstream report;
462 report.setf(std::ios::fixed);
463 report << std::setprecision(2)
464 << "Session pipeline report: mode=" << event.modeId
465 << ", sessionId=" << event.sessionId
466 << ", elapsedSeconds=" << event.elapsedSeconds
467 << ", frames=" << m_pipelineTimings.sampledFrames
468 << ", estimatedFps=" << estimatedFps
469 << ", avgFixedStepsPerFrame=" << avgFixedStepsPerFrame
470 << ", avgFrameBudget144Pct=" << avgBudget144Pct
471 << ", avgFrameBudget120Pct=" << avgBudget120Pct
472 << ", worstFrameBudget144Pct=" << worstBudget144Pct
473 << std::setprecision(3)
474 << ", avgFramePipelineMs=" << avgFramePipelineMs
475 << ", avgUiFrameTickMs=" << avgUiFrameTickMs
476 << ", avgModeUpdateMs=" << avgModeUpdateMs
477 << ", avgWorldSyncMs=" << avgWorldSyncMs
478 << ", avgUiFrameTickUs=" << (avgUiFrameTickMs * 1000.0)
479 << ", avgModeUpdateUs=" << (avgModeUpdateMs * 1000.0)
480 << ", avgWorldSyncUs=" << (avgWorldSyncMs * 1000.0)
481 << ", worstFramePipelineMs=" << m_pipelineTimings.framePipelineWorstMs
482 << ", worstUiFrameTickMs=" << m_pipelineTimings.uiFrameTickWorstMs
483 << ", worstModeUpdateMs=" << m_pipelineTimings.modeUpdateWorstMs
484 << ", worstWorldSyncMs=" << m_pipelineTimings.worldSyncWorstMs;
485
486 m_deps.logger.LogIf("DEBUG_PERF", core::LogLevel::Info, "perf",
487 report.str());
488 _ResetPipelineTimings();
489}

495void TrainingFlowController::_OnFrameTick(
496 const core::events::FrameTickEvent& tick) {
497 const auto framePipelineStart = std::chrono::steady_clock::now();
498
499 const auto uiStart = std::chrono::steady_clock::now();
500 m_deps.uiViewModel.UpdateFrameTick(tick);
501 const double uiFrameTickMs =
502 TimeUtils::ElapsedMilliseconds(uiStart, std::chrono::steady_clock::now());
503
504 double modeUpdateMs = 0.0;
505 if (m_state == State::Active) {
506 const auto modeStart = std::chrono::steady_clock::now();
507 for (std::uint32_t i = 0; i < tick.fixedSteps; ++i) {
508 m_deps.modeOrchestrator.Update(tick.fixedStepSeconds);
509 }
510 modeUpdateMs = TimeUtils::ElapsedMilliseconds(
511 modeStart, std::chrono::steady_clock::now());
512 }
513
514 double worldSyncMs = 0.0;
515 if (tick.fixedSteps > 0 || m_deps.world.RenderDirtyCount() > 0) {
516 const auto syncStart = std::chrono::steady_clock::now();
517 m_deps.worldRenderSync.Sync();
518 worldSyncMs = TimeUtils::ElapsedMilliseconds(
519 syncStart, std::chrono::steady_clock::now());
520 }
521
522 if (m_deps.modeOrchestrator.HasActiveMode()) {
523 _CollectPipelineSample(
524 m_deps.modeOrchestrator.ActiveSessionId(),
525 tick.fixedSteps,
526 TimeUtils::ElapsedMilliseconds(framePipelineStart,
527 std::chrono::steady_clock::now()),
528 uiFrameTickMs, modeUpdateMs, worldSyncMs);
529 }
530}

563void TrainingFlowController::_OnSessionSummary(
564 const core::events::SessionSummaryEvent& event) {
565 const core::events::SessionSummaryEvent summary = event;
566
567 QTimer::singleShot(0, m_deps.window, [this, summary]() {
568 _RefreshSessionHistory(summary.modeId);
569
570 const auto validation =
571 persistence::SessionHistory::ValidateSessionSummary(summary);
572 if (!validation.valid) {
573 m_deps.uiViewModel.SetLatestResultInvalid(summary.sessionId,
574 validation.reasonCode);
575 return;
576 }
577
578 if (!m_deps.profileManager || !m_deps.sessionHistory) {
579 return;
580 }
581
582 const std::int64_t profileId =
583 m_deps.profileManager->ActiveProfileId();
584 if (profileId <= 0 || summary.modeId.empty()) {
585 return;
586 }
587
588 const auto modePerformance =
589 m_deps.sessionHistory->GetModePerformanceSummary(profileId,
590 summary.modeId);
591 if (!modePerformance.has_value()) {
592 return;
593 }
594
595 std::vector<ui::UiViewModel::PerformanceMetricComparison>
596 metricComparisons;
597 std::vector<PerformanceClassifier::EvaluatedMetricComparison>
598 evaluatedComparisons;
599
600 const auto appendMetric =
601 [&](const std::string& metricId, double value,
602 const persistence::ModeMetricAggregate& aggregate,
603 PerformanceClassifier::MetricDirection direction,
604 double weight) {
605 const double best =
606 PerformanceClassifier::BestValueForDirection(aggregate,
607 direction);
608 const double worst =
609 PerformanceClassifier::WorstValueForDirection(aggregate,
610 direction);
611 const std::string tier =
612 PerformanceClassifier::ClassifyMetricTier(
613 modePerformance->sessionCount, value,
614 aggregate.average, best, worst, direction);
615
616 metricComparisons.push_back(
617 ui::UiViewModel::PerformanceMetricComparison{
618 metricId, tier, value, aggregate.average, best, worst});
619 evaluatedComparisons.push_back(
620 PerformanceClassifier::EvaluatedMetricComparison{tier,
621 weight});
622 };
623
624 appendMetric("score", static_cast<double>(summary.score),
625 modePerformance->score,
626 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.35);
627 appendMetric("hits", static_cast<double>(summary.hits),
628 modePerformance->hits,
629 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.10);
630 appendMetric("misses", static_cast<double>(summary.misses),
631 modePerformance->misses,
632 PerformanceClassifier::MetricDirection::LowerIsBetter, 0.10);
633 appendMetric("accuracyPercent", summary.accuracyPercent,
634 modePerformance->accuracyPercent,
635 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.25);
636 appendMetric("averageReactionTimeMs", summary.averageReactionTimeMs,
637 modePerformance->averageReactionTimeMs,
638 PerformanceClassifier::MetricDirection::LowerIsBetter, 0.15);
639 appendMetric("shotsPerSecond", summary.shotsPerSecond,
640 modePerformance->shotsPerSecond,
641 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.05);
642
643 const std::string overallTier =
644 PerformanceClassifier::ClassifyOverallPerformanceTier(
645 modePerformance->sessionCount, evaluatedComparisons);
646
647 m_deps.uiViewModel.SetLatestResultPerformanceBenchmark(
648 summary.sessionId, overallTier, modePerformance->sessionCount,
649 metricComparisons);
650 });
651}
