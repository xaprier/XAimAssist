39 std::vector<SessionRecord> result;
40 if (profileId <= 0 || limit <= 0) {
41 return result;
42 }
43
44 QSqlDatabase database = m_database.Connection();
45 if (!database.isOpen()) {
46 return result;

49 QSqlQuery query(database);
50 query.prepare(
51 "SELECT id, profile_id, runtime_session_id, mode_id, "
52 "started_at, stopped_at, elapsed_seconds, "
53 "score, shots_fired, hits, misses, targets_spawned, "
54 "targets_destroyed, accuracy_percent, "
55 "average_reaction_time_ms, shots_per_second, created_at "
56 "FROM session_history "
57 "WHERE profile_id = :profileId "
58 "ORDER BY created_at DESC "
59 "LIMIT :limit;");

107std::optional<ModePerformanceSummary>
108SessionHistory::GetModePerformanceSummary(std::int64_t profileId,
109 const std::string& modeId) const {
110 if (profileId <= 0 || modeId.empty()) {
111 return std::nullopt;
112 }
113
114 QSqlDatabase database = m_database.Connection();
115 if (!database.isOpen()) {
116 return std::nullopt;
117 }
118
119 QSqlQuery query(database);
120 query.prepare(
121 "SELECT COUNT(*), "
122 "AVG(score), MIN(score), MAX(score), "
123 "AVG(hits), MIN(hits), MAX(hits), "
124 "AVG(misses), MIN(misses), MAX(misses), "
125 "AVG(accuracy_percent), MIN(accuracy_percent), "
126 "MAX(accuracy_percent), "
127 "AVG(average_reaction_time_ms), "
128 "MIN(average_reaction_time_ms), "
129 "MAX(average_reaction_time_ms), "
130 "AVG(shots_per_second), MIN(shots_per_second), "
131 "MAX(shots_per_second) "
132 "FROM session_history "
133 "WHERE profile_id = :profileId AND mode_id = :modeId;");
134 query.bindValue(":profileId", static_cast<qlonglong>(profileId));
135 query.bindValue(":modeId", QString::fromStdString(modeId));
136
137 if (!query.exec() || !query.next()) {
138 return std::nullopt;
139 }
140
141 const auto sessionCount =
142 static_cast<std::uint64_t>(query.value(0).toLongLong());
143 if (sessionCount == 0) {
144 return std::nullopt;
145 }
146
147 ModePerformanceSummary summary;
148 summary.modeId = modeId;
149 summary.sessionCount = sessionCount;
150
151 const auto readAggregate = [&](int averageIndex, int minimumIndex,
152 int maximumIndex) -> ModeMetricAggregate {
153 ModeMetricAggregate aggregate;
154 aggregate.average = query.value(averageIndex).toDouble();
155 aggregate.minimum = query.value(minimumIndex).toDouble();
156 aggregate.maximum = query.value(maximumIndex).toDouble();
157 return aggregate;
158 };
159
160 summary.score = readAggregate(1, 2, 3);
161 summary.hits = readAggregate(4, 5, 6);
162 summary.misses = readAggregate(7, 8, 9);
163 summary.accuracyPercent = readAggregate(10, 11, 12);
164 summary.averageReactionTimeMs = readAggregate(13, 14, 15);
165 summary.shotsPerSecond = readAggregate(16, 17, 18);
166
167 return summary;
168}

170SessionValidationResult SessionHistory::ValidateSessionSummary(
171 const core::events::SessionSummaryEvent& event) {
172 if (event.averageReactionTimeMs <= 0.0) {
173 return SessionValidationResult{false, "reaction_time_zero"};
174 }
175
176 if (event.hits == 0) {
177 return SessionValidationResult{false, "hits_zero"};
178 }
179
180 if (event.score < 0) {
181 return SessionValidationResult{false, "score_negative"};
182 }
183
184 if (event.accuracyPercent <= ACCURACY_EXTREME_EPSILON ||
185 event.accuracyPercent > (100.0 + ACCURACY_EXTREME_EPSILON)) {
186 return SessionValidationResult{false, "accuracy_extreme"};
187 }
188
189 return SessionValidationResult{};
190}

192void SessionHistory::_HandleCoreEvent(const core::events::CoreEvent& event) {
193 if (const auto* sessionStartedEvent =

231void SessionHistory::_OnSessionSummary(
232 const core::events::SessionSummaryEvent& event) {
233 const std::int64_t profileId = m_profileManager.ActiveProfileId();
234 if (profileId <= 0) {
235 m_logger.Warning("persistence",
236 "Skipping session history Save: no active profile");
237 return;
238 }
239
240 RuntimeSessionContext context;
241 auto iterator = m_runtimeSessions.find(event.sessionId);
242 if (iterator != m_runtimeSessions.end()) {
243 context = iterator->second;
244 }
245
246 if (context.modeId.empty()) {
247 context.modeId = event.modeId;
248 }
249
250 if (!context.hasStarted) {
251 context.startedAt = std::chrono::system_clock::now();
252 context.hasStarted = true;
253 }
254
255 if (!context.hasStopped) {
256 context.stoppedAt = std::chrono::system_clock::now();
257 context.hasStopped = true;
258 }
259
260 if (context.stopReason == core::events::SessionStopReason::AbortedByUser) {
261 m_logger.Info("persistence",
262 "Skipping session history Save: training exited by user");
263 m_runtimeSessions.erase(event.sessionId);
264 return;
265 }
266
267 const SessionValidationResult validation = ValidateSessionSummary(event);
268 if (!validation.valid) {
269 m_logger.Warning("persistence",
270 "Skipping session history Save: invalid summary (reason=" +
271 validation.reasonCode + ")");
272 m_runtimeSessions.erase(event.sessionId);
273 return;
274 }
275
276 if (_PersistSessionSummary(profileId, event, context)) {
277 m_logger.Info("persistence", "Session summary persisted to SQLite history");
278 } else {
279 m_logger.Error("persistence",
280 "Failed to persist session summary to SQLite history");
281 }
282
283 m_runtimeSessions.erase(event.sessionId);
284}

286bool SessionHistory::_PersistSessionSummary(
287 std::int64_t profileId,
288 const core::events::SessionSummaryEvent& summaryEvent,
289 const RuntimeSessionContext& runtimeContext) {
290 QSqlDatabase database = m_database.Connection();
291 if (!database.isOpen()) {
292 return false;
293 }
294
295 QSqlQuery query(database);
296 query.prepare(
297 "INSERT INTO session_history ("
298 "profile_id, runtime_session_id, mode_id, started_at, "
299 "stopped_at, elapsed_seconds, score, "
300 "shots_fired, hits, misses, targets_spawned, "
301 "targets_destroyed, accuracy_percent, "
302 "average_reaction_time_ms, shots_per_second, created_at"
303 ") VALUES ("
304 ":profileId, :runtimeSessionId, :modeId, :startedAt, "
305 ":stoppedAt, :ElapsedSeconds, :score, "
306 ":shotsFired, :hits, :misses, :targetsSpawned, "
307 ":targetsDestroyed, :accuracyPercent, "
308 ":averageReactionTime, :shotsPerSecond, :createdAt"
309 ");");
310
311 query.bindValue(":profileId", static_cast<qlonglong>(profileId));
312 query.bindValue(":runtimeSessionId",
313 static_cast<qlonglong>(summaryEvent.sessionId));
314 query.bindValue(":modeId", QString::fromStdString(summaryEvent.modeId));
315 query.bindValue(":startedAt",
316 QString::fromStdString(_ToIsoUtc(runtimeContext.startedAt)));
317 query.bindValue(":stoppedAt",
318 QString::fromStdString(_ToIsoUtc(runtimeContext.stoppedAt)));
319 query.bindValue(":ElapsedSeconds", summaryEvent.elapsedSeconds);
320 query.bindValue(":score", static_cast<qlonglong>(summaryEvent.score));
321 query.bindValue(":shotsFired",
322 static_cast<qlonglong>(summaryEvent.shotsFired));
323 query.bindValue(":hits", static_cast<qlonglong>(summaryEvent.hits));
324 query.bindValue(":misses", static_cast<qlonglong>(summaryEvent.misses));
325 query.bindValue(":targetsSpawned",
326 static_cast<qlonglong>(summaryEvent.targetsSpawned));
327 query.bindValue(":targetsDestroyed",
328 static_cast<qlonglong>(summaryEvent.targetsDestroyed));
329 query.bindValue(":accuracyPercent", summaryEvent.accuracyPercent);
330 query.bindValue(":averageReactionTime", summaryEvent.averageReactionTimeMs);
331 query.bindValue(":shotsPerSecond", summaryEvent.shotsPerSecond);
332 query.bindValue(":createdAt", QString::fromStdString(_ToIsoUtc(
333 std::chrono::system_clock::now())));
334
335 return query.exec();
336}
