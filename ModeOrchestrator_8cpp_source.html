23bool ModeOrchestrator::StartMode(
24 const std::string& modeId, double durationOverrideSeconds,
25 double distanceOverrideUnits,
26 const std::unordered_map<std::string, double>& modeSettingOverrides) {
27 if (HasActiveMode()) {
28 StopMode();
29 }
30
31 auto mode =
32 m_registry.CreateMode(modeId, m_eventBus, m_logger, m_sceneCommandSink,
33 m_targetSystem, m_settingsProvider);
34 if (!mode) {
35 m_logger.Error("gameplay", "Requested mode is not registered");
36 return false;
37 }
38
39 m_activeSessionId = m_nextSessionId++;
40 m_activeModeId = modeId;
41 m_elapsedSeconds = 0.0;
42
43 const double defaultDuration = mode->Metadata().defaultDurationSeconds;
44 m_modeDurationSeconds =
45 durationOverrideSeconds > 0.0 ? durationOverrideSeconds : defaultDuration;
46
47 const double defaultDistance =
48 std::max(0.1, mode->Metadata().defaultDistanceUnits);
49 const double ConfiguredDistance =
50 distanceOverrideUnits > 0.0 ? distanceOverrideUnits : defaultDistance;
51
52 m_activeMode = std::move(mode);
53 m_activeMode->SetSessionId(m_activeSessionId);
54 m_activeMode->SetConfiguredDistance(std::max(0.1, ConfiguredDistance));
55 m_activeMode->ApplyModeSettings(modeSettingOverrides);
56 m_activeMode->OnStart();
57
58 const GameplayRuntimeSettings runtimeSettings =
59 m_settingsProvider.CurrentSettings();
60
61 m_eventBus.Publish(core::events::SessionStartedEvent{
62 m_activeSessionId, m_activeModeId, std::chrono::system_clock::now(),
63 std::max(0.1, ConfiguredDistance),
64 std::max(0.05, runtimeSettings.TargetRadius), modeSettingOverrides});
65
66 {
67 std::ostringstream debugStream;
68 debugStream << "Mode Start details: SessionId=" << m_activeSessionId
69 << ", mode=" << m_activeModeId
70 << ", durationSeconds=" << m_modeDurationSeconds
71 << ", ConfiguredDistance=" << ConfiguredDistance
72 << ", settingCount="
73 << static_cast<unsigned long long>(modeSettingOverrides.size());
74 m_logger.DebugIf("DEBUG_GAMEPLAY", "gameplay", debugStream.str());
75 }
76
77 m_logger.Info("gameplay", "Game mode started");
78 return true;
79}

81void ModeOrchestrator::Update(double dtSeconds) {
82 if (!HasActiveMode()) {
83 return;
84 }
85
86 const double safeDelta = std::max(0.0, dtSeconds);
87 m_elapsedSeconds += safeDelta;
88 m_activeMode->OnUpdate(safeDelta);
89 m_targetSystem.Update(safeDelta);
90
91 if (m_modeDurationSeconds > 0.0 &&
92 m_elapsedSeconds >= m_modeDurationSeconds) {
93 StopMode();
94 }
95}

97void ModeOrchestrator::StopMode(core::events::SessionStopReason stopReason) {
98 if (!HasActiveMode()) {
99 return;
100 }
101
102 m_activeMode->OnStop();
103 m_targetSystem.ClearSessionTargets(m_activeSessionId);
104
105 m_eventBus.Publish(core::events::SessionStoppedEvent{
106 m_activeSessionId, m_activeModeId, std::chrono::system_clock::now(),
107 m_elapsedSeconds, m_modeDurationSeconds, stopReason});
108
109 {
110 std::ostringstream debugStream;
111 debugStream << "Mode Stop details: SessionId=" << m_activeSessionId
112 << ", mode=" << m_activeModeId
113 << ", ElapsedSeconds=" << m_elapsedSeconds
114 << ", configuredDurationSeconds=" << m_modeDurationSeconds;
115 m_logger.DebugIf("DEBUG_GAMEPLAY", "gameplay", debugStream.str());
116 }
117
118 m_logger.Info("gameplay", "Game mode stopped");
119
120 m_activeMode.reset();
121 m_activeModeId.clear();
122 m_activeSessionId = 0;
123 m_elapsedSeconds = 0.0;
124 m_modeDurationSeconds = 0.0;
125}
