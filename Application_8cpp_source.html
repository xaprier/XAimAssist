342 auto collectSessionPipelineSample =
343 [&](std::uint64_t SessionId, std::uint32_t fixedSteps,
344 double framePipelineMs, double uiFrameTickMs, double modeUpdateMs,
345 double worldSyncMs) {
346 if (!sessionPipelineTimings.active ||
347 sessionPipelineTimings.SessionId != SessionId) {
348 return;
349 }
350
351 sessionPipelineTimings.sampledFrames += 1;
352 sessionPipelineTimings.sampledFixedSteps += fixedSteps;
353
354 sessionPipelineTimings.framePipelineMsSum += framePipelineMs;
355 sessionPipelineTimings.uiFrameTickMsSum += uiFrameTickMs;
356 sessionPipelineTimings.modeUpdateMsSum += modeUpdateMs;
357 sessionPipelineTimings.worldSyncMsSum += worldSyncMs;
358
359 sessionPipelineTimings.framePipelineWorstMs = std::max(
360 sessionPipelineTimings.framePipelineWorstMs, framePipelineMs);
361 sessionPipelineTimings.uiFrameTickWorstMs =
362 std::max(sessionPipelineTimings.uiFrameTickWorstMs, uiFrameTickMs);
363 sessionPipelineTimings.modeUpdateWorstMs =
364 std::max(sessionPipelineTimings.modeUpdateWorstMs, modeUpdateMs);
365 sessionPipelineTimings.worldSyncWorstMs =
366 std::max(sessionPipelineTimings.worldSyncWorstMs, worldSyncMs);
367 };

369 auto logSessionPipelineReport =
370 [&](const core::events::SessionStoppedEvent& event) {
371 if (!sessionPipelineTimings.active ||
372 sessionPipelineTimings.SessionId != event.sessionId) {
373 return;
374 }
375
376 if (sessionPipelineTimings.sampledFrames == 0) {
377 services.logger->LogIf(
378 "DEBUG_PERF", core::LogLevel::Info, "perf",
379 "Session pipeline report unavailable: no collected frames");
380 resetSessionPipelineTimings();
381 return;
382 }
383
384 const double sampledFrames =
385 static_cast<double>(sessionPipelineTimings.sampledFrames);
386 const double averageFramePipelineMs =
387 sessionPipelineTimings.framePipelineMsSum / sampledFrames;
388 const double averageUiFrameTickMs =
389 sessionPipelineTimings.uiFrameTickMsSum / sampledFrames;
390 const double averageModeUpdateMs =
391 sessionPipelineTimings.modeUpdateMsSum / sampledFrames;
392 const double averageWorldSyncMs =
393 sessionPipelineTimings.worldSyncMsSum / sampledFrames;
394 const double averageFixedStepsPerFrame =
395 static_cast<double>(sessionPipelineTimings.sampledFixedSteps) /
396 sampledFrames;
397 const double estimatedFps = event.elapsedSeconds > 0.0
398 ? sampledFrames / event.elapsedSeconds
399 : 0.0;
400 const double avgFrameBudgetUsage144Percent =
401 TARGET_FRAME_TIME_144_MS > 0.0
402 ? (averageFramePipelineMs * 100.0 / TARGET_FRAME_TIME_144_MS)
403 : 0.0;
404 const double avgFrameBudgetUsage120Percent =
405 TARGET_FRAME_TIME_120_MS > 0.0
406 ? (averageFramePipelineMs * 100.0 / TARGET_FRAME_TIME_120_MS)
407 : 0.0;
408 const double worstFrameBudgetUsage144Percent =
409 TARGET_FRAME_TIME_144_MS > 0.0
410 ? (sessionPipelineTimings.framePipelineWorstMs * 100.0 /
411 TARGET_FRAME_TIME_144_MS)
412 : 0.0;
413
414 std::ostringstream report;
415 report.setf(std::ios::fixed);
416 report << std::setprecision(2)
417 << "Session pipeline report: mode=" << event.modeId
418 << ", SessionId=" << event.sessionId
419 << ", ElapsedSeconds=" << event.elapsedSeconds
420 << ", frames=" << sessionPipelineTimings.sampledFrames
421 << ", estimatedFps=" << estimatedFps
422 << ", avgFixedStepsPerFrame=" << averageFixedStepsPerFrame
423 << ", avgFrameBudget144Pct=" << avgFrameBudgetUsage144Percent
424 << ", avgFrameBudget120Pct=" << avgFrameBudgetUsage120Percent
425 << ", worstFrameBudget144Pct=" << worstFrameBudgetUsage144Percent
426 << std::setprecision(3)
427 << ", avgFramePipelineMs=" << averageFramePipelineMs
428 << ", avgUiFrameTickMs=" << averageUiFrameTickMs
429 << ", avgModeUpdateMs=" << averageModeUpdateMs
430 << ", avgWorldSyncMs=" << averageWorldSyncMs
431 << ", avgUiFrameTickUs=" << (averageUiFrameTickMs * 1000.0)
432 << ", avgModeUpdateUs=" << (averageModeUpdateMs * 1000.0)
433 << ", avgWorldSyncUs=" << (averageWorldSyncMs * 1000.0)
434 << ", worstFramePipelineMs="
435 << sessionPipelineTimings.framePipelineWorstMs
436 << ", worstUiFrameTickMs="
437 << sessionPipelineTimings.uiFrameTickWorstMs
438 << ", worstModeUpdateMs="
439 << sessionPipelineTimings.modeUpdateWorstMs
440 << ", worstWorldSyncMs="
441 << sessionPipelineTimings.worldSyncWorstMs;
442
443 services.logger->LogIf("DEBUG_PERF", core::LogLevel::Info, "perf",
444 report.str());
445 resetSessionPipelineTimings();
446 };

674 const auto modeUpdateStart = std::chrono::steady_clock::now();

693 collectSessionPipelineSample(

732 if (trainingFlowState != TrainingFlowState::Menu) {
733 trainingFlowState = TrainingFlowState::Menu;
734 uiViewModel.ReturnToMainMenu();
735 setOverlayMousePassthrough(false);
736 services.inputManager->SetCaptureEnabled(false);
737 viewportWidget->hide();
738 ensureRuntimeLoopRunning();
739 }
740
741 return;
742 }
743
744 if (const auto* sessionSummaryEvent =
745 std::get_if<core::events::SessionSummaryEvent>(&event)) {
746 const core::events::SessionSummaryEvent summary =
747 *sessionSummaryEvent;
748 QTimer::singleShot(0, &window, [&, summary]() {
749 refreshSessionHistory(summary.modeId);
750
751 const auto validation =
752 persistence::SessionHistory::ValidateSessionSummary(summary);
753 if (!validation.valid) {
754 uiViewModel.SetLatestResultInvalid(summary.sessionId,
755 validation.reasonCode);
756 return;
757 }
758
759 if (!services.profileManager || !services.sessionHistory) {
760 return;
761 }
762
763 const std::int64_t ActiveProfileId =
764 services.profileManager->ActiveProfileId();
765 if (ActiveProfileId <= 0 || summary.modeId.empty()) {
766 return;
767 }
768
769 const auto modePerformance =
770 services.sessionHistory->GetModePerformanceSummary(
771 ActiveProfileId, summary.modeId);
772 if (!modePerformance.has_value()) {
773 return;
774 }
775
776 std::vector<ui::UiViewModel::PerformanceMetricComparison>
777 metricComparisons;
778 std::vector<PerformanceClassifier::EvaluatedMetricComparison> evaluatedComparisons;
779
780 const auto appendMetricComparison =
781 [&](const std::string& metricId, double metricValue,
782 const persistence::ModeMetricAggregate& aggregate,
783 PerformanceClassifier::MetricDirection direction, double weight) {
784 const double bestValue =
785 PerformanceClassifier::BestValueForDirection(aggregate, direction);
786 const double worstValue =
787 PerformanceClassifier::WorstValueForDirection(aggregate, direction);
788 const std::string metricTier = PerformanceClassifier::ClassifyMetricTier(
789 modePerformance->sessionCount, metricValue,
790 aggregate.average, bestValue, worstValue, direction);
791
792 metricComparisons.push_back(
793 ui::UiViewModel::PerformanceMetricComparison{
794 metricId, metricTier, metricValue, aggregate.average,
795 bestValue, worstValue});
796 evaluatedComparisons.push_back(
797 PerformanceClassifier::EvaluatedMetricComparison{metricTier, weight});
798 };
799
800 appendMetricComparison("score", static_cast<double>(summary.score),
801 modePerformance->score,
802 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.35);
803 appendMetricComparison("hits", static_cast<double>(summary.hits),
804 modePerformance->hits,
805 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.10);
806 appendMetricComparison(
807 "misses", static_cast<double>(summary.misses),
808 modePerformance->misses, PerformanceClassifier::MetricDirection::LowerIsBetter, 0.10);
809 appendMetricComparison("accuracyPercent", summary.accuracyPercent,
810 modePerformance->accuracyPercent,
811 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.25);
812 appendMetricComparison("averageReactionTimeMs",
813 summary.averageReactionTimeMs,
814 modePerformance->averageReactionTimeMs,
815 PerformanceClassifier::MetricDirection::LowerIsBetter, 0.15);
816 appendMetricComparison("shotsPerSecond", summary.shotsPerSecond,
817 modePerformance->shotsPerSecond,
818 PerformanceClassifier::MetricDirection::HigherIsBetter, 0.05);
819
820 const std::string overallTier = PerformanceClassifier::ClassifyOverallPerformanceTier(
821 modePerformance->sessionCount, evaluatedComparisons);
822
823 uiViewModel.SetLatestResultPerformanceBenchmark(
824 summary.sessionId, overallTier, modePerformance->sessionCount,
825 metricComparisons);
826 });
827 }
828 });
829
830 worldRenderSync.Sync();
831 ensureRuntimeLoopRunning();
832 setOverlayMousePassthrough(false);
833
834 services.eventBus->Publish(core::events::ApplicationLifecycleEvent{
835 core::events::ApplicationLifecycleState::Running});
836 services.logger->Info("app", "Runtime loop started");
837
838 QObject::connect(&qtApplication, &QCoreApplication::aboutToQuit, [&]() {
839 countdownTimer.stop();
840 stopRuntimeLoop();
841
842 if (modeOrchestrator.HasActiveMode()) {
843 modeOrchestrator.StopMode();
844 }
845
846 services.inputManager->SetCaptureEnabled(false);
847 services.inputManager->DetachViewport();
848 services.eventBus->Unsubscribe(frameUpdateSubscription);
849 services.eventBus->Unsubscribe(uiEventsSubscription);
850 worldRenderSync.Reset();
851 services.engine->Shutdown();
852 services.eventBus->Publish(core::events::ApplicationLifecycleEvent{
853 core::events::ApplicationLifecycleState::Stopping});
854 services.logger->Info("app", "XAimAssist Shutdown sequence completed");
855 });
856
857 window.show();
858 return qtApplication.exec();
859}
