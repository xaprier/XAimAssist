43std::string ClassifyMetricTier(
44 std::uint64_t sessionCount,
45 double value,
46 double averageValue,
47 double bestValue,
48 double worstValue,
49 MetricDirection direction) {
50 // First session has no comparison
51 if (sessionCount <= 1) {
52 return "first_result";
53 }
54
55 if (direction == MetricDirection::HigherIsBetter) {
56 // For metrics where higher is better
57 if (value >= bestValue - COMPARISON_EPSILON) {
58 return "best_result";
59 }
60 if (value <= worstValue + COMPARISON_EPSILON) {
61 return "worst_result";
62 }

64 return "above_average";

67 return "below_average";

77 return "worst_result";

88std::string ClassifyOverallPerformanceTier(
89 std::uint64_t sessionCount,
90 const std::vector<EvaluatedMetricComparison>& metricComparisons) {
91 // First session
92 if (sessionCount <= 1) {
93 return "first_result";
94 }
95
96 // No metrics to evaluate
97 if (metricComparisons.empty()) {
98 return "average";
99 }
100
101 // Calculate weighted average of tier scores
102 double weightedSum = 0.0;
103 double totalWeight = 0.0;
104
105 for (const auto& comparison : metricComparisons) {
106 if (comparison.weight <= 0.0) {
107 continue;
108 }
109 weightedSum += comparison.weight * TierScore(comparison.tier);
110 totalWeight += comparison.weight;
111 }
112
113 if (totalWeight <= 0.0) {
114 return "average";
115 }
116
117 // Normalize and classify
118 const double normalizedScore = weightedSum / totalWeight;
119
120 // Thresholds for tier classification
121 if (normalizedScore >= 1.4) {
122 return "best_result";
123 }
124 if (normalizedScore <= -1.4) {
125 return "worst_result";
126 }
127 if (normalizedScore >= 0.35) {
128 return "above_average";
129 }
130 if (normalizedScore <= -0.35) {
131 return "below_average";
132 }
133
134 return "average";
135}
