38// Analytic ray-sphere intersection; returns nearest positive t or -1.
39double intersectRaySphere(const std::array<double, 3>& origin,
40 const std::array<double, 3>& direction,
41 const std::array<double, 3>& center, double radius) {
42 const auto oc = subtract(origin, center);
43 const double b = dot(oc, direction);
44 const double c = dot(oc, oc) - radius * radius;
45 const double discriminant = b * b - c;
46 if (discriminant < 0.0) {
47 return -1.0;
48 }
49
50 const double root = std::sqrt(discriminant);
51 const double nearDistance = -b - root;
52 const double farDistance = -b + root;
53
54 if (nearDistance > 0.0001) {
55 return nearDistance;
56 }
57
58 if (farDistance > 0.0001) {
59 return farDistance;
60 }
61
62 return -1.0;
63}
