9SoundSystem::SoundSystem(core::EventBus& eventBus) : m_eventBus(eventBus) {
10 m_hitSubId = m_eventBus.Subscribe(
11 [this](const core::events::CoreEvent& event) {
12 if (const auto* e =
13 std::get_if<core::events::ShotHitEvent>(&event)) {
14 _OnShotHit(*e);
15 }
16 });
17
18 m_missSubId = m_eventBus.Subscribe(
19 [this](const core::events::CoreEvent& event) {
20 if (const auto* e =
21 std::get_if<core::events::ShotMissEvent>(&event)) {
22 _OnShotMiss(*e);
23 }
24 });
25
26 ApplySettings(persistence::SoundPreferences{});
27}

34void SoundSystem::ApplySettings(const persistence::SoundPreferences& prefs) {
35 m_prefs = prefs;
36
37 const QUrl hitUrl = _HitSoundUrl(prefs.hitSound);
38 if (m_hitEffect.source() != hitUrl) {
39 m_hitEffect.setSource(hitUrl);
40 }
41 m_hitEffect.setVolume(static_cast<float>(prefs.volume));
42
43 const QUrl missUrl = _MissSoundUrl(prefs.missSound);
44 if (m_missEffect.source() != missUrl) {
45 m_missEffect.setSource(missUrl);
46 }
47 m_missEffect.setVolume(static_cast<float>(prefs.volume));
48}

50void SoundSystem::_OnShotHit(const core::events::ShotHitEvent&) {
