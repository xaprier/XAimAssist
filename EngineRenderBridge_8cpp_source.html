29bool EngineRenderBridge::UpdateSphere(

31 const world::RenderComponent& render) {
32 auto cacheIterator = m_renderStateCache.find(objectId);
33 if (cacheIterator == m_renderStateCache.end()) {

36 cacheIterator = m_renderStateCache.find(objectId);
37 }
38
39 RenderStateCache& cachedState = cacheIterator->second;
40
41 // Diff against cached state to avoid redundant GPU calls
42 const bool positionChanged = cachedState.position != transform.position;
43 const bool colorChanged = cachedState.color != render.color;
44 const bool opacityChanged =
45 std::abs(cachedState.opacity - render.opacity) > 0.0001;
46
47 if (!positionChanged && !colorChanged && !opacityChanged) {
48 return true;
49 }
50
51 bool updated = true;
52 if (positionChanged) {

55 }
56

59 }

62 updated = m_engine.SetObjectOpacity(objectId, render.opacity) && updated;
63 }
64
65 if (!updated) {
66 return false;
67 }

70 cachedState.color = render.color;
71 cachedState.opacity = render.opacity;
