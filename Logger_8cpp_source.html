35bool parseBoolFlagValue(const char* value) {
36 if (value == nullptr) {
37 return false;
38 }
39
40 std::string normalized(value);
41 for (char& character : normalized) {
42 character =
43 static_cast<char>(std::tolower(static_cast<unsigned char>(character)));
44 }
45
46 return normalized == "1" || normalized == "true" || normalized == "on" ||
47 normalized == "yes" || normalized == "y";

108bool Logger::DebugFlagEnabled(std::string_view flagName) {
109 if (flagName.empty()) {
110 return false;
111 }
112
113 const std::string normalizedFlagName(flagName);
114
115 std::lock_guard<std::mutex> lock(g_debugFlagMutex);
116 const auto cached = g_debugFlagCache.find(normalizedFlagName);
117 if (cached != g_debugFlagCache.end()) {
118 return cached->second;
119 }
120
121 const bool enabled = readDebugFlagNoCache(normalizedFlagName);
122 g_debugFlagCache.emplace(normalizedFlagName, enabled);
123 return enabled;
124}
