36std::int64_t ProfileManager::CreateProfile(const std::string& profileName) {
37 QSqlDatabase database = m_database.Connection();
38 if (!database.isOpen()) {
39 return 0;
40 }
41
42 QSqlQuery query(database);
43 query.prepare(
44 "INSERT INTO Profiles(name, created_at, updated_at, is_active) "
45 "VALUES(:name, :createdAt, :updatedAt, 0);");
46
47 const QString timestamp = QString::fromStdString(_NowIsoUtc());
48 query.bindValue(":name", QString::fromStdString(profileName));
49 query.bindValue(":createdAt", timestamp);
50 query.bindValue(":updatedAt", timestamp);
51
52 if (!query.exec()) {
53 m_logger.Error("persistence", "Failed to create profile");
54 return 0;
55 }
56
57 return query.lastInsertId().toLongLong();
58}

60bool ProfileManager::SetActiveProfile(std::int64_t profileId) {
61 if (profileId <= 0) {
62 return false;
63 }
64
65 QSqlDatabase database = m_database.Connection();
66 if (!database.isOpen()) {
67 return false;
68 }
69
70 if (!database.transaction()) {
71 return false;
72 }
73
74 QSqlQuery clearActive(database);
75 if (!clearActive.exec("UPDATE Profiles SET is_active = 0;")) {
76 database.rollback();
77 return false;
78 }
79
80 QSqlQuery setActive(database);
81 setActive.prepare(
82 "UPDATE Profiles "
83 "SET is_active = 1, updated_at = :updatedAt "
84 "WHERE id = :id;");
85 setActive.bindValue(":updatedAt", QString::fromStdString(_NowIsoUtc()));
86 setActive.bindValue(":id", static_cast<qlonglong>(profileId));
87
88 if (!setActive.exec() || setActive.numRowsAffected() <= 0) {
89 database.rollback();
90 return false;
91 }
92
93 return database.commit();
94}

135std::vector<ProfileRecord> ProfileManager::Profiles() const {
136 std::vector<ProfileRecord> result;
137
138 QSqlDatabase database = m_database.Connection();
139 if (!database.isOpen()) {
140 return result;
141 }
142
143 QSqlQuery query(database);
144 if (!query.exec("SELECT id, name, created_at, updated_at, is_active "
145 "FROM Profiles "
146 "ORDER BY id ASC;")) {
147 return result;
148 }
149
150 while (query.next()) {
151 ProfileRecord record;
152 record.id = query.value(0).toLongLong();
153 record.name = query.value(1).toString().toStdString();
154 record.createdAtIso = query.value(2).toString().toStdString();
155 record.updatedAtIso = query.value(3).toString().toStdString();
156 record.active = query.value(4).toInt() != 0;
157 result.push_back(std::move(record));
158 }
159
160 return result;
161}
