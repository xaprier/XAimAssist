41void StaticSphereMode::OnStart() {
42 m_previousSpawnOffset.reset();

44 _SpawnTargetAtCurrentSlot();

47 GetEventBus().Subscribe([this](const core::events::CoreEvent& event) {

82void StaticSphereMode::_SpawnTargetAtCurrentSlot() {
83 const GameplayRuntimeSettings runtimeSettings =
84 GetSettingsProvider().CurrentSettings();
85
86 TargetDefinition Definition;
87 Definition.type = TargetType::Static;
88 Definition.radius = runtimeSettings.TargetRadius;
89
90 const double spawnDistanceX = std::max(0.1, ConfiguredDistance());
91 const double spawnRangeY = TARGET_Y_MAX - TARGET_Y_MIN;
92 const double spawnRangeZ = TARGET_Z_MAX - TARGET_Z_MIN;
93 const double maxSpawnOffsetDistance =
94 std::sqrt(spawnRangeY * spawnRangeY + spawnRangeZ * spawnRangeZ);
95 const double requestedMinimumSpawnOffsetDistance = spawnDistanceX / 5.0;
96 const double cappedMinimumSpawnOffsetDistance =
97 std::min(requestedMinimumSpawnOffsetDistance,
98 maxSpawnOffsetDistance * MIN_SPAWN_DISTANCE_CAP_FACTOR);
99 const double _MinimumSpawnOffsetDistance =
100 std::clamp(cappedMinimumSpawnOffsetDistance, 0.0, maxSpawnOffsetDistance);
101
102 const auto spawnOffset = _RandomSpawnOffset(_MinimumSpawnOffsetDistance);
103 Definition.initialPosition = {spawnDistanceX, spawnOffset[0], spawnOffset[1]};
104 Definition.color = runtimeSettings.targetColor;
105 Definition.collidable = true;
106
107 m_activeTargetId = GetTargetSystem().SpawnTarget(SessionId(), Definition);
108 if (m_activeTargetId == 0) {
109 GetLogger().Warning("gameplay", "StaticSphereMode failed to spawn target");
110 return;
111 }
112
113 m_previousSpawnOffset = spawnOffset;
114}

116std::array<double, 2>
117StaticSphereMode::_RandomSpawnOffset(double minimumDistance) {
118 std::uniform_real_distribution<double> yDistribution(TARGET_Y_MIN,
119 TARGET_Y_MAX);
120 std::uniform_real_distribution<double> zDistribution(TARGET_Z_MIN,
121 TARGET_Z_MAX);
122
123 std::array<double, 2> bestOffset = {
124 yDistribution(m_randomEngine),
125 zDistribution(m_randomEngine),
126 };
127
128 if (!m_previousSpawnOffset.has_value()) {
129 return bestOffset;
130 }
131
132 double bestDistance = _PlanarDistance(bestOffset, *m_previousSpawnOffset);
133 if (bestDistance >= minimumDistance) {
134 return bestOffset;
135 }
136
137 for (std::size_t attempt = 0; attempt < MAX_RANDOM_SAMPLE_ATTEMPTS;
138 ++attempt) {
139 const std::array<double, 2> candidateOffset = {
140 yDistribution(m_randomEngine),
141 zDistribution(m_randomEngine),
142 };
143
144 const double candidateDistance =
145 _PlanarDistance(candidateOffset, *m_previousSpawnOffset);
146 if (candidateDistance >= minimumDistance) {
147 return candidateOffset;
148 }
149
150 if (candidateDistance > bestDistance) {
151 bestDistance = candidateDistance;
152 bestOffset = candidateOffset;
153 }
154 }
155
156 return bestOffset;
157}

159double StaticSphereMode::_PlanarDistance(const std::array<double, 2>& left,
160 const std::array<double, 2>& right) {
161 const double deltaY = left[0] - right[0];
162 const double deltaZ = left[1] - right[1];
163 return std::sqrt(deltaY * deltaY + deltaZ * deltaZ);
164}
