127bool InputManager::eventFilter(QObject* watched, QEvent* event) {
128 if (m_viewportWidget.isNull()) {
129 m_logger.Warning("input",
130 "Event filter called but viewport is null - detaching");
131 DetachViewport();
132 return QObject::eventFilter(watched, event);
133 }
134
135 if (watched != m_viewportWidget.data()) {
136 return QObject::eventFilter(watched, event);
137 }
138
139 switch (event->type()) {
140 case QEvent::MouseButtonPress: {
141 const auto* mouseEvent = static_cast<QMouseEvent*>(event);
142
143 if (!m_captureEnabled) {
144 if (mouseEvent->button() == Qt::LeftButton && m_sceneClickHandler) {
145 if (m_sceneClickHandler()) {
146 return true;
147 }
148 }
149
150 return QObject::eventFilter(watched, event);
151 }
152
153 _HandleMouseButton(true, mouseEvent->button());
154 return true;
155 }
156 case QEvent::MouseButtonRelease: {
157 if (!m_captureEnabled) {
158 return QObject::eventFilter(watched, event);
159 }
160
161 _HandleMouseButton(false, static_cast<QMouseEvent*>(event)->button());
162 return true;
163 }
164 case QEvent::MouseMove:
165 if (!m_captureEnabled) {
166 return QObject::eventFilter(watched, event);
167 }
168
169 _HandleMouseMove(*static_cast<QMouseEvent*>(event));
170 return true;
171 case QEvent::KeyPress: {
172 const auto* keyEvent = static_cast<QKeyEvent*>(event);
173 if (keyEvent->key() == Qt::Key_Escape) {
174 if (m_escapeHandler) {
175 m_escapeHandler();
176 } else if (m_captureEnabled) {
177 SetCaptureEnabled(false);
178 }
179
180 return true;
181 }
182
183 if (!m_captureEnabled) {
184 return QObject::eventFilter(watched, event);
185 }
186
187 break;
188 }
189 default:
190 if (!m_captureEnabled) {
191 return QObject::eventFilter(watched, event);
192 }
193
194 break;
195 }
196
197 return QObject::eventFilter(watched, event);
198}

211void InputManager::_HandleMouseMove(const QMouseEvent& event) {
212 if (!_ValidateViewport("_HandleMouseMove")) {
213 return;
214 }
215
216 if (m_recentering) {
217 m_recentering = false;
218 return;
219 }
220
221 double deltaX = 0.0;
222 double deltaY = 0.0;
223
224 if (m_rawInputEnabled) {
225 const QPointF localCenter = m_viewportWidget->rect().center();
226 const QPointF mouseLocalPosition = event.position();
227 deltaX = mouseLocalPosition.x() - localCenter.x();
228 deltaY = mouseLocalPosition.y() - localCenter.y();
229
230 _CenterCursorInViewport();
231 } else {
232 const QPointF mouseGlobalPosition = event.globalPosition();
233 if (!m_hasLastPointerPosition) {
234 m_lastPointerGlobalPosition = mouseGlobalPosition;
235 m_hasLastPointerPosition = true;
236 return;
237 }
238
239 deltaX = mouseGlobalPosition.x() - m_lastPointerGlobalPosition.x();
240 deltaY = mouseGlobalPosition.y() - m_lastPointerGlobalPosition.y();
241 m_lastPointerGlobalPosition = mouseGlobalPosition;
242 }
243
244 if (std::abs(deltaX) < 0.0001 && std::abs(deltaY) < 0.0001) {
245 return;
246 }
247
248 if (!m_lookHandler) {
249 return;
250 }
251
252 const auto [yawDeltaDegrees, pitchDeltaDegrees] =
253 m_sensitivityModel.ComputeLookDeltaDegrees(deltaX, deltaY);
254
255 m_lookHandler(LookAction{yawDeltaDegrees, pitchDeltaDegrees});
256}
