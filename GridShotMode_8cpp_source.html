99void GridShotMode::OnStart() {
100 m_gridRowCount = clampedAxisCount(
101 static_cast<int>(std::lround(ModeSettingValue(
102 "grid_rows", static_cast<double>(GRID_DEFAULT_ROW_COUNT)))),
103 GRID_DEFAULT_ROW_COUNT);
104 m_gridColumnCount = clampedAxisCount(
105 static_cast<int>(std::lround(ModeSettingValue(
106 "grid_columns", static_cast<double>(GRID_DEFAULT_COLUMN_COUNT)))),
107 GRID_DEFAULT_COLUMN_COUNT);
108
109 const std::size_t gridCellCount = m_gridRowCount * m_gridColumnCount;
110 const std::size_t maxActiveTargetCount =
111 gridCellCount > 1 ? (gridCellCount - 1) : 1;
112
113 const int requestedActiveTargetCount = static_cast<int>(std::lround(
114 ModeSettingValue("active_target_count",
115 static_cast<double>(GRID_DEFAULT_ACTIVE_TARGET_COUNT))));
116 m_activeTargetCount = static_cast<std::size_t>(std::clamp(
117 requestedActiveTargetCount, 1,
118 static_cast<int>(std::max<std::size_t>(1, maxActiveTargetCount))));
119
120 m_activeTargetCellById.clear();
121
122 for (std::size_t targetIndex = 0; targetIndex < m_activeTargetCount;
123 ++targetIndex) {
124 _SpawnTargetAtRandomAvailableCell();
125 }
126
127 m_shotHitSubscriptionId =
128 GetEventBus().Subscribe([this](const core::events::CoreEvent& event) {
129 const auto* shotHitEvent =
130 std::get_if<core::events::ShotHitEvent>(&event);
131 if (shotHitEvent == nullptr) {
132 return;
133 }
134
135 if (shotHitEvent->sessionId != SessionId()) {
136 return;
137 }
138
139 const auto destroyedTargetIterator =
140 m_activeTargetCellById.find(shotHitEvent->targetId);
141 if (destroyedTargetIterator == m_activeTargetCellById.end()) {
142 return;
143 }
144
145 const std::size_t destroyedCellIndex = destroyedTargetIterator->second;
146 m_activeTargetCellById.erase(destroyedTargetIterator);
147 _SpawnTargetAtRandomAvailableCell(destroyedCellIndex);
148 });
149
150 GetLogger().Info("gameplay", "GridShotMode started");
151}

165void GridShotMode::_SpawnTargetAtRandomAvailableCell(
166 std::optional<std::size_t> excludedCellIndex) {
167 const auto availableCellIndex = _RandomAvailableCellIndex(excludedCellIndex);
168 if (!availableCellIndex.has_value()) {
169 GetLogger().Warning("gameplay",
170 "GridShotMode has no available cell for target spawn");
171 return;
172 }
173
174 const GameplayRuntimeSettings runtimeSettings =
175 GetSettingsProvider().CurrentSettings();
176
177 TargetDefinition Definition;
178 Definition.type = TargetType::Static;
179 Definition.radius = runtimeSettings.TargetRadius;
180 Definition.initialPosition = _WorldPositionForCell(*availableCellIndex);
181 Definition.color = runtimeSettings.targetColor;
182 Definition.collidable = true;
183
184 const std::uint64_t TargetId =
185 GetTargetSystem().SpawnTarget(SessionId(), Definition);
186 if (TargetId == 0) {
187 GetLogger().Warning("gameplay", "GridShotMode failed to spawn target");
188 return;
189 }
190
191 m_activeTargetCellById.emplace(TargetId, *availableCellIndex);
192}

194std::optional<std::size_t> GridShotMode::_RandomAvailableCellIndex(
195 std::optional<std::size_t> excludedCellIndex) {
196 const std::size_t gridCellCount = m_gridRowCount * m_gridColumnCount;
197 if (m_activeTargetCellById.size() >= gridCellCount) {
198 return std::nullopt;
199 }
200
201 std::vector<std::size_t> availableCells;
202 availableCells.reserve(gridCellCount - m_activeTargetCellById.size());
203
204 for (std::size_t cellIndex = 0; cellIndex < gridCellCount; ++cellIndex) {
205 if (excludedCellIndex.has_value() &&
206 cellIndex == excludedCellIndex.value()) {
207 continue;
208 }
209
210 if (!_IsCellOccupied(cellIndex)) {
211 availableCells.push_back(cellIndex);
212 }
213 }
214
215 if (availableCells.empty()) {
216 return std::nullopt;
217 }
218
219 std::uniform_int_distribution<std::size_t> indexDistribution(
220 0, availableCells.size() - 1);
221 return availableCells[indexDistribution(m_randomEngine)];
222}

224std::array<double, 3>
225GridShotMode::_WorldPositionForCell(std::size_t cellIndex) const {
226 const std::size_t rowIndex = cellIndex / m_gridColumnCount;
227 const std::size_t columnIndex = cellIndex % m_gridColumnCount;
228
229 const double spawnDistanceX = std::max(0.1, ConfiguredDistance());
230 const double cellSpacing = std::max(0.1, spawnDistanceX / 10.0);
231
232 const double rowCenter = (static_cast<double>(m_gridRowCount) - 1.0) * 0.5;
233 const double columnCenter =
234 (static_cast<double>(m_gridColumnCount) - 1.0) * 0.5;
235
236 const double offsetY =
237 (static_cast<double>(columnIndex) - columnCenter) * cellSpacing;
238 const double offsetZ =
239 (rowCenter - static_cast<double>(rowIndex)) * cellSpacing;
240
241 return {spawnDistanceX, offsetY, offsetZ};
242}

244bool GridShotMode::_IsCellOccupied(std::size_t cellIndex) const {
245 return std::any_of(
246 m_activeTargetCellById.begin(), m_activeTargetCellById.end(),
247 [cellIndex](const auto& entry) { return entry.second == cellIndex; });
248}
