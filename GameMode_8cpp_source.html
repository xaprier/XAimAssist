34void GameMode::ApplyModeSettings(
35 const std::unordered_map<std::string, double>& requestedSettings) {
36 m_modeSettings.clear();
37
38 for (const auto& setting : m_metadata.settings) {
39 const auto requestedIterator = requestedSettings.find(setting.key);
40 const double requestedValue = requestedIterator != requestedSettings.end()
41 ? requestedIterator->second
42 : setting.defaultValue;
43
44 const double low = std::min(setting.minValue, setting.maxValue);
45 const double high = std::max(setting.minValue, setting.maxValue);
46
47 double clampedValue = std::clamp(requestedValue, low, high);
48 if (setting.integerOnly) {
49 clampedValue = std::round(clampedValue);
50 }
51
52 m_modeSettings.emplace(setting.key, clampedValue);
53 }
54}
