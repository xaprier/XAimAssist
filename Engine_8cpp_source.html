24QSurfaceFormat Engine::RecommendedSurfaceFormat() {
25 QSurfaceFormat fmt = QVTKOpenGLNativeWidget::defaultFormat();
26
27 // OpenGL 4.1 Core is the highest version available on macOS and is broadly
28 // supported on NVIDIA/AMD/Intel across all target platforms.
29 fmt.setVersion(4, 1);
30 fmt.setProfile(QSurfaceFormat::CoreProfile);
31
32 // swap_interval=0: immediate present — no vertical sync, no added latency.
33 // Competitive aim trainers always disable VSync for minimum input-to-pixel
34 // delay. Users who want VSync can enable it via the driver control panel.
35 fmt.setSwapInterval(0);
36
37 // SwapBehavior::DoubleBuffer: one back buffer only.
38 // TripleBuffer adds ~1 frame of latency on some GL implementations.
39 fmt.setSwapBehavior(QSurfaceFormat::DoubleBuffer);
40
41 // Depth/stencil: 24-bit depth is sufficient for the scene distances used
42 // in an aim trainer; 8-bit stencil reserved for future effects.
43 fmt.setDepthBufferSize(24);
44 fmt.setStencilBufferSize(8);
45
46 // No MSAA at the surface level — antialiasing (if desired) is better
47 // handled in a post-process pass that doesn't bloat the default framebuffer.
48 fmt.setSamples(0);
49
50 return fmt;
51}

78Engine::SceneObjectId Engine::SpawnSphere(const SphereSpawnRequest& request) {
79 if (!m_initialized) {
80 return 0;
81 }
82
83 auto sphereSource = vtkSmartPointer<vtkSphereSource>::New();
84 sphereSource->SetRadius(request.radius);
85 sphereSource->SetThetaResolution(request.thetaResolution);
86 sphereSource->SetPhiResolution(request.phiResolution);
87
88 auto mapper = vtkSmartPointer<vtkPolyDataMapper>::New();
89 mapper->SetInputConnection(sphereSource->GetOutputPort());
90
91 auto actor = vtkSmartPointer<vtkActor>::New();
92 actor->SetMapper(mapper);
93 actor->SetPosition(request.position[0], request.position[1],
94 request.position[2]);
95 actor->GetProperty()->SetOpacity(std::clamp(request.opacity, 0.0, 1.0));
96 actor->GetProperty()->SetColor(request.color[0], request.color[1],
97 request.color[2]);
98
99 return m_scene.AddActor(actor);
100}

145bool Engine::_InitializeViewport(QVTKOpenGLNativeWidget& viewportWidget) {
146 if (m_initialized) {
147 return true;
148 }
149
150 if (!m_renderContext.Initialize(viewportWidget)) {
151 m_logger.Error("engine", "Failed to Initialize Render context");
152 return false;
153 }
154
155 m_scene.SetRenderer(m_renderContext.Renderer());
156 m_cameraBackend.SetRenderer(m_renderContext.Renderer());
157
158 SetBackgroundColor({0.05, 0.08, 0.16});
159 SetCameraPose(CameraPose{});
160
161 m_initialized = true;
162 _SubscribeToCoreEvents();
163
164 m_logger.Info("engine", "Engine initialization complete");
165 return true;
